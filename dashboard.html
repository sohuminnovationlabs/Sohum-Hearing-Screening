const AudQueueScreen = ({allRows,onBack,scriptUrl,audiologists=[],onReassigned}) => { const [expandedAud, setExpandedAud] = useState(null); const [reassigning, setReassigning] = useState(null); const [targetAud, setTargetAud] = useState(''); const [msg, setMsg] = useState(''); const s = useMemo(() => calcStats(allRows), [allRows]); const queueMap = s.audQueueMap; const allAuds = Object.keys(queueMap).sort(); const availableAuds = audiologists.length ? audiologists : allAuds.map(username => ({ username, name: username })); const handleReassign = (screeningId, currentAud) => { setReassigning({ screeningId, currentAud }); setTargetAud(''); setMsg(''); }; const submitReassign = () => { if (!targetAud) { setMsg('Select a target audiologist.'); return; } if (DATA_MODE === 'offline') { setMsg('Offline mode - reassignment not persisted.'); setReassigning(null); return; } fetch(scriptUrl, { method:'POST', headers:{'Content-Type':'text/plain'}, body: JSON.stringify({ action:'reassignCase', screeningId:reassigning.screeningId, newAudiologist:targetAud, adminUsername:'admin' }) }).then(r=>r.json()).then(d=>{ if (d.status === 'success') { setMsg(`Reassigned to ${targetAud}`); if (onReassigned) onReassigned(); } else { setMsg(`Error: ${d.message}`); } setReassigning(null); setTargetAud(''); }).catch(()=>{ setMsg('Network error'); setReassigning(null); }); }; return h('div',{className:'fade-in',style:{minHeight:'100vh',paddingBottom:60}}, h(AppHeader,{title:'Audiologist Queue',topLabel:'Admin View',subtitle:'Case management and reassignment'}), h('div',{style:{padding:'14px 14px 0'}}, h(BackBtn,{onClick:onBack}), h('div',{style:{display:'flex',gap:8,marginBottom:12}}, h(KpiCard,{value:s.revPending, label:'Pending', color:s.revPending>0?C.orange:C.green, bg:s.revPending>0?C.orangeL:C.greenL}), h(KpiCard,{value:s.revPhotoReq, label:'Photo Req.', color:s.revPhotoReq>0?C.orange:C.slate400}), h(KpiCard,{value:s.revCompleted,label:'Completed', color:C.green}) ), msg && h('div',{style:{background:C.greenL,border:`1px solid ${C.green}`,borderRadius:10,padding:'8px 12px',fontSize:12,color:C.greenD,marginBottom:10}},msg), allAuds.length === 0 && h('div',{style:{textAlign:'center',color:C.slate400,fontSize:13,padding:'32px 0'}},'No audiologist assignments in the selected data range.'), allAuds.map(aud => { const q = queueMap[aud]; const isExpanded = expandedAud === aud; return h(Card,{key:aud,style:{marginBottom:10}}, h('div',{onClick:()=>setExpandedAud(isExpanded?null:aud),style:{cursor:'pointer'}}, h('div',{style:{display:'flex',justifyContent:'space-between',alignItems:'center'}}, h('div',{}, h('div',{style:{fontSize:14,fontWeight:600,color:C.slate800}},aud), h('div',{style:{display:'flex',gap:4,marginTop:4,flexWrap:'wrap'}}, q.pending>0 && h(Chip,{label:`${q.pending} pending`, bg:C.yellowL,fg:C.yellow}), q.photoReq>0 && h(Chip,{label:`${q.photoReq} photo req`,bg:C.orangeL,fg:C.orange}), q.photoRec>0 && h(Chip,{label:`${q.photoRec} photo rec`,bg:C.tealL, fg:C.teal}), q.completed>0 && h(Chip,{label:`${q.completed} done`, bg:C.greenL, fg:C.greenD}), q.urgent>0 && h(Chip,{label:`${q.urgent} urgent`, bg:C.redL, fg:C.red}) ) ), h('svg',{xmlns:'http://www.w3.org/2000/svg',width:16,height:16,viewBox:'0 0 24 24',fill:'none',stroke:C.slate400,strokeWidth:2}, h('path',{d:isExpanded?'m18 15-6-6-6 6':'m6 9 6 6 6-6'})) ) ), isExpanded && h('div',{style:{marginTop:12}}, h(Divider,{my:0}), h('div',{style:{marginTop:10}}, q.cases.sort((a,b) => { if(a.urgent && !b.urgent) return -1; if(!a.urgent && b.urgent) return 1; const statusOrder = {pending:0,photo_received:1,photo_requested:2,completed:3}; const aRank = statusOrder[a.reviewStatus] ?? 9; const bRank = statusOrder[b.reviewStatus] ?? 9; if (aRank !== bRank) return aRank - bRank; return new Date(b.assignedAt || b.ts || 0) - new Date(a.assignedAt || a.ts || 0); }).map(c => h('div',{key:c.screenId,style:{borderRadius:10,border:`1px solid ${C.slate200}`,padding:'10px 12px',marginBottom:8,background:c.urgent?'#fff7ed':C.white}}, h('div',{style:{display:'flex',justifyContent:'space-between',alignItems:'flex-start'}}, h('div',{style:{flex:1}}, h('div',{style:{display:'flex',gap:6,alignItems:'center',flexWrap:'wrap'}}, h('span',{style:{fontSize:12,fontWeight:600,color:C.slate800}},c.babyName||'Unknown'), c.urgent && h(Chip,{label:'URGENT',bg:C.redL,fg:C.red}), h(StatusPill,{status:c.reviewStatus}) ), h('div',{style:{fontSize:10,color:C.slate400,marginTop:3}},`${c.screenId} · ${c.hospital} · ${c.device}`), c.assignedAt && h('div',{style:{fontSize:10,color:C.slate400}},`Assigned: ${fmtDate(new Date(c.assignedAt))}`) ), c.reviewStatus === 'pending' && h('button',{ onClick:()=>handleReassign(c.screenId, aud), style:{background:C.accentL,color:C.accent,border:'none',borderRadius:8,padding:'4px 10px',fontSize:11,fontWeight:600,cursor:'pointer',flexShrink:0,fontFamily:'inherit',marginLeft:8} },'Reassign') ) )) ) ) ); }), reassigning && h('div',{ style:{ position:'fixed',inset:0,background:'rgba(15,23,42,0.45)',zIndex:99999, display:'flex',alignItems:'center',justifyContent:'center',padding:18 } }, h('div',{style:{background:C.white,borderRadius:16,padding:16,width:'100%',maxWidth:420,boxShadow:'0 18px 50px rgba(15,23,42,0.25)'}}, h('div',{style:{fontSize:16,fontWeight:700,color:C.slate800,marginBottom:4}},'Reassign Case'), h('div',{style:{fontSize:11,color:C.slate500,marginBottom:12}},reassigning.screeningId), h('div',{style:{display:'flex',flexDirection:'column',gap:8,maxHeight:300,overflowY:'auto',marginBottom:14}}, availableAuds .filter(a => a.username !== reassigning.currentAud) .map(a => h('label',{key:a.username,style:{display:'flex',alignItems:'center',gap:10,padding:'9px 10px',border:`1px solid ${targetAud===a.username?C.primary:C.slate200}`,borderRadius:10,cursor:'pointer',background:targetAud===a.username?C.accentL:C.white}}, h('input',{type:'radio',name:'targetAudiologist',checked:targetAud===a.username,onChange:()=>setTargetAud(a.username)}), h('div',{}, h('div',{style:{fontSize:13,fontWeight:600,color:C.slate800}},a.name || a.username), h('div',{style:{fontSize:10,color:C.slate400}},a.username) ) )) ), h('div',{style:{display:'flex',gap:8}}, h('button',{onClick:submitReassign,style:{flex:1,background:C.primary,color:'#fff',border:'none',borderRadius:10,padding:'10px',fontSize:13,fontWeight:700,cursor:'pointer',fontFamily:'inherit'}},'Confirm'), h('button',{onClick:()=>{setReassigning(null);setTargetAud('');},style:{flex:1,background:C.slate200,color:C.slate600,border:'none',borderRadius:10,padding:'10px',fontSize:13,fontWeight:700,cursor:'pointer',fontFamily:'inherit'}},'Cancel') ) ) ) ) ); };
